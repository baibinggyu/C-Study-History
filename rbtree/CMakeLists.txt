cmake_minimum_required(VERSION 3.25.1)
project(RBTree VERSION 1.0)
SET(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(USE_RBTree "RBTre is use?" on)
if(USE_RBTree)
    add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/RBTree)
    LIST(APPEND EXTERA_LIB RBTree)
    List(APPEND EXTERA_INCLUDE ${CMAKE_SOURCE_DIR}/RBTree)
    message(STATUS "RBTree_lib is ON")
endif()

file(GLOB SRC_DIR "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")
file(GLOB INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/include")
add_executable(main ${SRC_DIR})
target_link_directories(main PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(main PRIVATE ${EXTERA_LIB})
target_include_directories(main PRIVATE ${INCLUDE_DIR} ${EXTERA_INCLUDE})
